abes-profile-gateway.ts
SECURE PROXY
// Query authorized student profile through the institutional gateway
import { ErpGateway } from '@abes/erp-security';

const gateway = new ErpGateway({
  timeoutMs: 15000,
  sessionScope: 'institutional'
});

export async function getStudentProfile(admissionNumber: string) {
  const profile = await gateway.fetchProfile({
    id: admissionNumber,
    format: 'image/jpeg'
  });
  return profile.stream();
}
